#include <stdio.h>

int main()
{
    int n, rev = 0, digit;
    printf("Enter a number: ");
    scanf("%d", &n);
    while (n != 0) {
        digit = n % 10;               /* take the last digit */
        rev = rev * 10 + digit;       /* add it to the reversed number */
        n = n / 10;                   /* remove the last digit */
    }
    printf("Reversed number = %d\n", rev);
    return 0;
}

    printf("Enter a number: ");
    scanf("%d", &n);
    /* count the digits */
    temp = n;
    while (temp != 0) {
        digits++;
        temp = temp / 10;
    }
    /* add each digit raised to the power of number of digits */
    temp = n;
    while (temp != 0) {
        digit = temp % 10;
        power = 1;
        for (i = 0; i < digits; i++)
            power = power * digit;
        sum = sum + power;
        temp = temp / 10;
    }
    if (sum == n)
        printf("%d is an Armstrong number\n", n);
    else
        printf("%d is not an Armstrong number\n", n);
    return 0;
}
